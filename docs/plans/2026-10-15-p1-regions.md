# P1 Region Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace v1's Python "symbols" with regions: an exact, deterministic, local partition of every file into leasable, versionable spans. This includes language packs, IDs, lifecycle, the shifted map for broken parses, code edges, `regionMerge`, the lock format and `loom regions`. Gate: L−1.

**Architecture:** `packages/indexer/src/regions/` is a pure library. It takes text in and returns regions out, with no graph or daemon. A file goes through tiers (whole-file checks → markers → language pack → JSON → Markdown → agent-defined → file). Each tier emits *pieces*, and `assemble` turns pieces into an exact cover with stable IDs and hashes. Everything downstream (P2 graph, P3 leases, P4 conflicts) consumes `FileRegions`, `Region`, `diffRegions`, `buildEdges`, `regionMerge` and the lock helpers.

**Tech Stack:** TypeScript 5.6 (ESM, NodeNext), web-tree-sitter 0.22.6 + tree-sitter-wasms 0.1.13, `diff` 5.2.0 (`diffLines`), git CLI (`merge-file`), vitest 2.1.8.

**Spec:** `docs/specs/2026-10-15-loom-v2-design.md` §3 (regions), §10 (L−1), §10.1 (ledger).

## Global Constraints

- Offsets inside the engine are JS string (UTF-16) indices. The lock stores UTF-8 byte offsets (§3.4).
- `hash` = sha1 of the region's UTF-8 bytes. `masked` = sha1 with the defining name token replaced by `\u0000`.
- IDs are `<path>::<qname>::<kind>#<n>`. No line numbers or offsets, ever (§3.3).
- A parse with any ERROR or MISSING node never produces a new map (§3.6).
- Whole-file: binary/non-UTF-8, over 1 MB, average line length over 300, lockfiles, `@generated` in the first 5 lines (§3.7).
- web-tree-sitter 0.22.6 API (smoke-verified 2026-10-15):
  - `node.hasError` and `node.isNamed` are **properties**;
  - queries are `language.query(src).captures(node)`;
  - Python `return_type` text has no `->`, TS `return_type` text starts with `:`.
- Bundled CJS: `createRequire(import.meta.url ?? __filename)`, and the CLI marks `web-tree-sitter` and `tree-sitter-wasms` external and depends on them (smoke-verified).
- No commits or pushes before the replay finishes (~16:30 IST, 2026-10-15). Commit steps run after that, in task order, with real timestamps.

## Review Focus

1. **Two statements on one line** (`a = 1; b = 2`, or two TS methods on one line): pieces overlap after line extension. Expected: merged into the earlier region with a warning, and exact cover still holds. Pinned in Task 2.
2. **File with a BOM or CRLF line endings:** offsets and line extension must still give an exact cover. Pinned in Task 4.
3. **Empty file and whitespace-only file:** zero regions, or one gap, never a crash. Pinned in Task 4.
4. **Insertion at offset 0 in a degraded file:** the shifted map must still start at 0. Pinned in Task 5.
5. **regionMerge where both sides append a new function at the end:** both are kept, in a deterministic order, and the result parses. Pinned in Task 8.

```ts
// packages/indexer/src/regions/types.ts
export type RegionKind = 'definition' | 'container' | 'imports' | 'gap' | 'key' | 'section' | 'file';

export interface Region {
  id: string;            // <path>::<qname>::<kind>#<n>
  kind: RegionKind;
  qname: string;
  start: number;         // UTF-16 index, inclusive
  end: number;           // UTF-16 index, exclusive
  hash: string;          // sha1 of the UTF-8 bytes
  masked: string;        // sha1 with the defining name token replaced by \u0000 (= hash when there is no name)
  sig: string | null;    // contract text: name + params (+ " -> " return type)
}

export interface CallSite { from: string; name: string }                                       // region id → callee text
export interface ImportBinding { from: string; local: string; module: string; name: string | null } // region id of the import

export interface FileRegions {
  path: string;
  pack: string;          // "python@1", "json@1", "markdown@1", "defined@1", "file", "binary"
  degraded: boolean;
  binary?: boolean;
  text: string;
  regions: Region[];
  warnings: string[];
  touched?: string[];    // degraded maps only: ids whose text changed since the last clean map
  calls?: CallSite[];
  imports?: ImportBinding[];
}

export interface Piece {
  start: number; end: number; kind: RegionKind; qname: string;
  nameStart?: number; nameEnd?: number; sig: string | null;
}

export interface RegionizeOpts {
  lastClean?: FileRegions;
  defined?: Array<{ start: number; end: number; name: string }>;
  generated?: boolean;
}
```

```ts
// packages/indexer/src/regions/text.ts
import { createHash } from 'node:crypto';

export const sha1 = (s: string | Buffer): string => createHash('sha1').update(s).digest('hex');

export function decodeUtf8(b: Buffer): string | null {
  try { return new TextDecoder('utf-8', { fatal: true }).decode(b); } catch { return null; }
}

/** Start of the line that contains index i. */
export const lineStart = (text: string, i: number): number => (i <= 0 ? 0 : text.lastIndexOf('\n', i - 1) + 1);

/** End of the line that contains index i-1 (a node's exclusive end), newline included. */
export function lineEnd(text: string, i: number): number {
  if (i <= 0 || text[i - 1] === '\n') return i;
  const j = text.indexOf('\n', i);
  return j === -1 ? text.length : j + 1;
}

export const onlySpaceBefore = (text: string, i: number): boolean => /^[ \t]*$/.test(text.slice(lineStart(text, i), i));

export const squash = (s: string): string => s.replace(/\s+/g, ' ').replace(/\( /g, '(').replace(/ \)/g, ')').trim();
```

```ts
// packages/indexer/src/regions/invariants.ts
import { sha1 } from './text.js';
import type { FileRegions } from './types.js';

/** §3.2 invariants 1–2 plus hash and id integrity. Determinism and locality are property-tested. */
export function checkInvariants(fr: FileRegions): string[] {
  if (fr.binary) return [];
  const out: string[] = [];
  const ids = new Set<string>();
  let at = 0;
  for (const r of fr.regions) {
    if (r.start !== at) out.push(`${r.id}: starts at ${r.start}, expected ${at}`);
    if (r.end <= r.start) out.push(`${r.id}: empty`);
    else if (sha1(fr.text.slice(r.start, r.end)) !== r.hash) out.push(`${r.id}: hash mismatch`);
    if (ids.has(r.id)) out.push(`${r.id}: duplicate id`);
    ids.add(r.id);
    at = Math.max(at, r.end);
  }
  if (at !== fr.text.length) out.push(`cover ends at ${at}, text length ${fr.text.length}`);
  return out;
}
```

```ts
// packages/indexer/src/regions/pack.ts
import { createRequire } from 'node:module';
import Parser from 'web-tree-sitter';
import python from './packs/python.json' with { type: 'json' };
import { lineEnd, lineStart, onlySpaceBefore, squash } from './text.js';
import type { Piece } from './types.js';

declare const __filename: string;
// import.meta.url is undefined inside the CLI's CJS bundle (smoke 2026-10-15)
const require = createRequire(import.meta.url ?? __filename);

export interface Pack {
  name: string; version: number; extensions: string[]; grammar: string; comment: string;
  importReader: 'python' | 'ecmascript';
  wrappers: Record<string, string>; definitions: string[]; assignments: string[];
  containers: Record<string, string>; members: string[]; imports: string[]; calls: string; params: string[];
}
type Node = Parser.SyntaxNode;

export const PACKS: Pack[] = [python as Pack];
export const packFor = (path: string): Pack | undefined => PACKS.find((p) => p.extensions.some((e) => path.endsWith(e)));

let ready: Promise<void> | null = null;
const langs = new Map<string, Promise<Parser.Language>>();
const parsers = new Map<string, Parser>();

export async function language(grammar: string): Promise<Parser.Language> {
  ready ??= Parser.init();
  await ready;
  let l = langs.get(grammar);
  if (!l) { l = Parser.Language.load(require.resolve(`tree-sitter-wasms/out/${grammar}`)); langs.set(grammar, l); }
  return l;
}

export async function parse(grammar: string, text: string): Promise<Parser.Tree> {
  const lang = await language(grammar);
  let p = parsers.get(grammar);
  if (!p) { p = new Parser(); p.setLanguage(lang); parsers.set(grammar, p); }
  return p.parse(text);
}

/** web-tree-sitter 0.22.6 exposes hasError as a property; tolerate the method form of other versions. */
export function hasError(n: Node): boolean {
  const h = (n as unknown as { hasError: boolean | (() => boolean) }).hasError;
  return typeof h === 'function' ? h.call(n) : h;
}

const unwrap = (pack: Pack, n: Node): Node => {
  const f = pack.wrappers[n.type];
  return (f && n.childForFieldName(f)) || n;
};

function nameNode(pack: Pack, def: Node): Node | null {
  if (pack.assignments.includes(def.type)) {
    const c = def.namedChildren[0];
    const n = c?.type === 'assignment' ? c.childForFieldName('left') : c?.type === 'variable_declarator' ? c.childForFieldName('name') : null;
    return n?.type === 'identifier' ? n : null;
  }
  if (!pack.definitions.includes(def.type) && !pack.members.includes(def.type)) return null;
  return def.childForFieldName('name');
}

function sigOf(pack: Pack, def: Node, name: string): string | null {
  const holder = pack.assignments.includes(def.type) ? def.namedChildren[0]?.childForFieldName('value') ?? null : def;
  if (!holder) return null;
  let params: Node | null = null;
  for (const f of pack.params) if ((params = holder.childForFieldName(f))) break;
  if (!params) return null;
  const ret = holder.childForFieldName('return_type');
  const r = ret ? squash(ret.text).replace(/^:\s*/, '') : '';
  return `${name}${squash(params.text)}${r ? ` -> ${r}` : ''}`;
}

/** Line start of n, extended up over comment lines directly above it (no blank line between). */
function extStart(text: string, n: Node): number {
  let first = n;
  for (let p = n.previousNamedSibling; p && p.type === 'comment' && p.endPosition.row === first.startPosition.row - 1
    && onlySpaceBefore(text, p.startIndex); p = p.previousNamedSibling) first = p;
  return lineStart(text, first.startIndex);
}

const isMember = (pack: Pack, c: Node): boolean => pack.members.includes(c.type) && nameNode(pack, unwrap(pack, c)) !== null;

export function packPieces(pack: Pack, root: Node, text: string): Piece[] {
  const out: Piece[] = [];
  walk(pack, root.namedChildren, text, '', true, out);
  return out;
}

function walk(pack: Pack, nodes: Node[], text: string, prefix: string, top: boolean, out: Piece[]): void {
  for (const n of nodes) {
    if (n.type === 'comment') continue;
    if (top && pack.imports.includes(n.type)) {
      const start = extStart(text, n), end = lineEnd(text, n.endIndex);
      const prev = out.at(-1);
      if (prev?.kind === 'imports' && prev.end === start) prev.end = end;
      else out.push({ start, end, kind: 'imports', qname: 'imports', sig: null });
      continue;
    }
    if (!top && !isMember(pack, n)) continue;
    const def = unwrap(pack, n);
    const nm = nameNode(pack, def);
    if (!nm) continue;
    const qname = prefix + nm.text;
    const start = extStart(text, n), end = lineEnd(text, n.endIndex);
    const piece: Piece = { start, end, kind: 'definition', qname, nameStart: nm.startIndex, nameEnd: nm.endIndex, sig: sigOf(pack, def, nm.text) };
    const field = pack.containers[def.type];
    const body = field ? def.childForFieldName(field) : null;
    const members = body ? body.namedChildren.filter((c) => isMember(pack, c)) : [];
    if (members.length) {
      const headerEnd = extStart(text, members[0]);
      if (headerEnd > start) {
        out.push({ ...piece, kind: 'container', end: headerEnd });
        walk(pack, body!.namedChildren, text, `${qname}.`, false, out);
```

```ts
// packages/indexer/src/regions/assemble.ts
import { sha1 } from './text.js';
import type { Piece, Region } from './types.js';

/** Sort pieces, merge overlaps (with a warning), fill gaps, number duplicates, hash. Gives an exact cover. */
export function assemble(path: string, text: string, pieces: Piece[], warnings: string[]): Region[] {
  const sorted = [...pieces].sort((a, b) => a.start - b.start || a.end - b.end);
  const kept: Piece[] = [];
  for (const p of sorted) {
    const prev = kept.at(-1);
    if (prev && p.start < prev.end) {
      warnings.push(`overlap: ${p.qname} merged into ${prev.qname}`);
      prev.end = Math.max(prev.end, p.end);
      continue;
    }
    if (p.end > p.start) kept.push({ ...p });
  }
  const spans: Piece[] = [];
  let at = 0;
  let prevQ: string | null = null;
  const gap = (start: number, end: number): Piece =>
    ({ start, end, kind: 'gap', qname: prevQ === null ? 'gap@start' : `gap@after:${prevQ}`, sig: null });
  for (const p of kept) {
    if (p.start > at) spans.push(gap(at, p.start));
    spans.push(p);
    at = p.end;
    prevQ = p.qname;
  }
  if (at < text.length) spans.push(gap(at, text.length));
  const seen = new Map<string, number>();
  return spans.map((p) => {
    const key = `${p.qname}::${p.kind}`;
    const n = seen.get(key) ?? 0;
    seen.set(key, n + 1);
    const body = text.slice(p.start, p.end);
    const masked = p.nameStart === undefined ? body
      : body.slice(0, p.nameStart - p.start) + '\u0000' + body.slice(p.nameEnd! - p.start);
    return { id: `${path}::${key}#${n}`, kind: p.kind, qname: p.qname, start: p.start, end: p.end, hash: sha1(body), masked: sha1(masked), sig: p.sig };
  });
}
```

```ts
// packages/indexer/src/regions/regionize.ts
import { assemble } from './assemble.js';
import { hasError, packFor, packPieces, parse } from './pack.js';
import { decodeUtf8, sha1 } from './text.js';
import type { FileRegions, Piece, RegionizeOpts } from './types.js';

export async function regionize(path: string, input: string | Buffer, _opts: RegionizeOpts = {}): Promise<FileRegions> {
  const warnings: string[] = [];
  const text = typeof input === 'string' ? input : decodeUtf8(input);
  if (text === null) return binaryFile(path, input as Buffer);
  const pack = packFor(path);
  if (!pack) return wholeFile(path, text, 'file', false, warnings);
  const packName = `${pack.name}@${pack.version}`;
  const tree = await parse(pack.grammar, text);
  try {
    if (hasError(tree.rootNode)) return wholeFile(path, text, packName, true, warnings);
    return { path, pack: packName, degraded: false, text, warnings, regions: assemble(path, text, packPieces(pack, tree.rootNode, text), warnings) };
  } finally {
    tree.delete();
  }
}

export function wholeFile(path: string, text: string, pack: string, degraded: boolean, warnings: string[]): FileRegions {
  const pieces: Piece[] = text.length ? [{ start: 0, end: text.length, kind: 'file', qname: 'file', sig: null }] : [];
  return { path, pack, degraded, text, warnings, regions: assemble(path, text, pieces, warnings) };
}

function binaryFile(path: string, buf: Buffer): FileRegions {
  const h = sha1(buf);
  return {
    path, pack: 'binary', degraded: false, binary: true, text: '', warnings: ['whole-file: binary'],
    regions: buf.length ? [{ id: `${path}::file::file#0`, kind: 'file', qname: 'file', start: 0, end: buf.length, hash: h, masked: h, sig: null }] : [],
  };
}
```

```ts
// packages/indexer/src/regions/tiers.ts
import type Parser from 'web-tree-sitter';
import { lineEnd } from './text.js';
import type { Piece, RegionizeOpts } from './types.js';

const MB = 1 << 20;
const LOCKFILE = /(^|\/)(package-lock\.json|pnpm-lock\.yaml|yarn\.lock|poetry\.lock|[^/]+\.lock)$/;

export function wholeFileReason(path: string, text: string, bytes: number, opts: RegionizeOpts): string | null {
  if (bytes > MB) return 'over 1 MB';
  if (LOCKFILE.test(path)) return 'lockfile';
  if (opts.generated || /@generated/.test(text.split('\n', 5).join('\n'))) return 'generated';
  const lines = Math.max(1, text.split('\n').length - (text.endsWith('\n') ? 1 : 0));
  if (text.length / lines > 300) return 'minified';
  return null;
}

export function COMMENT_SYNTAX(path: string): [string, string] | null {
  if (/\.(py|pyi|sh|ya?ml|toml|rb)$/.test(path)) return ['# ', ''];
  if (/\.([cm]?[jt]sx?|go|rs|java|c|h|cpp|swift|kt)$/.test(path)) return ['// ', ''];
  if (/\.(md|markdown|html?)$/i.test(path)) return ['<!-- ', ' -->'];
  if (/\.sql$/.test(path)) return ['-- ', ''];
  return null;
}

export interface Block { start: number; end: number; name: string }
const OPEN = /^[ \t]*(?:#|\/\/|--|<!--)[ \t]*loom:region[ \t]+(\S+)/;
const CLOSE = /^[ \t]*(?:#|\/\/|--|<!--)[ \t]*loom:end\b/;

export function markerBlocks(text: string, warnings: string[]): Block[] {
  const blocks: Block[] = [];
  let open: { start: number; name: string; line: number; bad: boolean } | null = null;
  let off = 0;
  const lines = text.split('\n');
  lines.forEach((line, i) => {
    const start = off;
    off += line.length + (i < lines.length - 1 ? 1 : 0);
    const o = OPEN.exec(line);
    if (o) {
      if (open) { warnings.push(`marker: nested loom:region at line ${i + 1}; ${open.name} ignored`); open.bad = true; }
      else open = { start, name: o[1].replace(/-->$/, ''), line: i + 1, bad: false };
    } else if (CLOSE.test(line)) {
      if (!open) warnings.push(`marker: loom:end without loom:region at line ${i + 1}`);
      else { if (!open.bad) blocks.push({ start: open.start, end: off, name: open.name }); open = null; }
    }
  });
  if (open) warnings.push(`marker: unclosed loom:region ${(open as { name: string }).name} at line ${(open as { line: number }).line}`);
  return blocks;
}

/** A block replaces the pieces inside it; one that cuts through a piece is ignored (gaps may be split). */
export function overlayMarkers(pieces: Piece[], blocks: Block[], warnings: string[]): Piece[] {
  let out = pieces;
  for (const b of blocks) {
    const cut = out.find((p) => p.start < b.end && p.end > b.start && (p.start < b.start || p.end > b.end));
    if (cut) { warnings.push(`marker: ${b.name} cuts ${cut.kind} ${cut.qname}; ignored`); continue; }
    out = [...out.filter((p) => !(p.start >= b.start && p.end <= b.end)), { start: b.start, end: b.end, kind: 'section', qname: b.name, sig: null }];
  }
  return out;
}

export function jsonPieces(root: Parser.SyntaxNode): Piece[] | null {
  const obj = root.namedChildren.find((c) => c.type !== 'comment');
  if (obj?.type !== 'object') return null;
  return obj.namedChildren.filter((c) => c.type === 'pair').map((p) => {
    const k = p.childForFieldName('key')!;
    const next = p.nextSibling;
    let name: string;
    try { name = String(JSON.parse(k.text)); } catch { name = k.text; }
    return { start: p.startIndex, end: next?.type === ',' ? next.endIndex : p.endIndex, kind: 'key' as const, qname: name, nameStart: k.startIndex, nameEnd: k.endIndex, sig: null };
  });
}

export function markdownPieces(text: string): Piece[] {
  const out: Piece[] = [];
  const stack: string[] = [];
  let fence: string | null = null;
  let off = 0;
  const lines = text.split('\n');
  lines.forEach((line, i) => {
    const start = off;
    off += line.length + (i < lines.length - 1 ? 1 : 0);
    const f = /^ {0,3}(`{3,}|~{3,})/.exec(line);
    if (fence) { if (f && f[1][0] === fence[0] && f[1].length >= fence.length) fence = null; return; }
    if (f) { fence = f[1]; return; }
    const h = /^ {0,3}(#{1,6})(?:[ \t]+(.*?))?[ \t]*#*[ \t]*\r?$/.exec(line);
    if (!h) return;
    const level = h[1].length;
    stack.length = level;
    stack[level - 1] = (h[2] ?? '').trim();
    const qname = stack.slice(0, level).filter((s) => s).join('/') || '(untitled)';
    const prev = out.at(-1);
    if (prev) prev.end = start;
    out.push({ start, end: text.length, kind: 'section', qname, sig: null });
  });
  return out;
}

export function definedPieces(text: string, defined: Array<{ start: number; end: number; name: string }>, warnings: string[]): Piece[] | null {
  const ds = [...defined].sort((a, b) => a.start - b.start);
  for (let i = 0; i < ds.length; i++) {
    const d = ds[i];
    const why = !/^\S+$/.test(d.name) ? `bad name "${d.name}"` : d.start < 0 || d.end > text.length || d.end <= d.start ? `${d.name} out of bounds`
      : i > 0 && d.start < ds[i - 1].end ? `${d.name} overlaps ${ds[i - 1].name}` : null;
    if (why) { warnings.push(`defined: ${why}; ignored`); return null; }
  }
  return ds.map((d) => ({ start: d.start, end: d.end, kind: 'section' as const, qname: d.name, sig: null }));
}

export { lineEnd };
```

```ts
// packages/indexer/src/regions/regionize.ts  (full replacement)
import { assemble } from './assemble.js';
import { hasError, packFor, packPieces, parse } from './pack.js';
import { decodeUtf8, sha1 } from './text.js';
import { definedPieces, jsonPieces, markdownPieces, markerBlocks, overlayMarkers, wholeFileReason } from './tiers.js';
import type { FileRegions, Piece, RegionizeOpts } from './types.js';

const JSON_GRAMMAR = 'tree-sitter-json.wasm';

export async function regionize(path: string, input: string | Buffer, opts: RegionizeOpts = {}): Promise<FileRegions> {
  const warnings: string[] = [];
  const text = typeof input === 'string' ? input : decodeUtf8(input);
  if (text === null) return binaryFile(path, input as Buffer);
  const why = wholeFileReason(path, text, typeof input === 'string' ? Buffer.byteLength(input) : input.length, opts);
  if (why) { warnings.push(`whole-file: ${why}`); return wholeFile(path, text, 'file', false, warnings); }
  const blocks = markerBlocks(text, warnings);
  let pieces: Piece[] | null = null;
  let packName = 'file';
  const pack = packFor(path);
  if (pack) {
    packName = `${pack.name}@${pack.version}`;
    const tree = await parse(pack.grammar, text);
    try {
      if (hasError(tree.rootNode)) return wholeFile(path, text, packName, true, warnings);
      pieces = packPieces(pack, tree.rootNode, text);
    } finally {
      tree.delete();
    }
  } else if (/\.json$/i.test(path)) {
    const tree = await parse(JSON_GRAMMAR, text);
    try {
      if (!hasError(tree.rootNode) && (pieces = jsonPieces(tree.rootNode))) packName = 'json@1';
    } finally {
      tree.delete();
    }
  } else if (/\.(md|markdown)$/i.test(path)) {
    pieces = markdownPieces(text);
    packName = 'markdown@1';
  }
  if (pieces === null && opts.defined?.length && (pieces = definedPieces(text, opts.defined, warnings))) packName = 'defined@1';
  if (pieces === null && !blocks.length) return wholeFile(path, text, 'file', false, warnings);
  const regions = assemble(path, text, overlayMarkers(pieces ?? [], blocks, warnings), warnings);
  return { path, pack: packName, degraded: false, text, warnings, regions };
}

export function wholeFile(path: string, text: string, pack: string, degraded: boolean, warnings: string[]): FileRegions {
  const pieces: Piece[] = text.length ? [{ start: 0, end: text.length, kind: 'file', qname: 'file', sig: null }] : [];
  return { path, pack, degraded, text, warnings, regions: assemble(path, text, pieces, warnings) };
}

function binaryFile(path: string, buf: Buffer): FileRegions {
  const h = sha1(buf);
  return {
    path, pack: 'binary', degraded: false, binary: true, text: '', warnings: ['whole-file: binary'],
    regions: buf.length ? [{ id: `${path}::file::file#0`, kind: 'file', qname: 'file', start: 0, end: buf.length, hash: h, masked: h, sig: null }] : [],
  };
}
```

```ts
// packages/indexer/src/regions/shift.ts
import { diffLines } from 'diff';
import { sha1 } from './text.js';
import type { FileRegions, Region } from './types.js';

/**
 * §3.6: carry the last clean map onto broken text. Each region start is mapped through a line diff:
 * - kept text moves with its offset;
 * - a start inside a deletion moves to the deletion point;
 * - insertions belong to the region before them.
 * Monotonic, so exact cover holds. Emptied regions are dropped.
 */
export function shiftMap(last: FileRegions, text: string, warnings: string[]): FileRegions {
  const starts = last.regions.map((r) => r.start);
  const mapped: number[] = new Array(starts.length);
  let i = 0, o = 0, n = 0;
  for (const part of diffLines(last.text, text)) {
    const len = part.value.length;
    if (part.added) { n += len; continue; }
    while (i < starts.length && starts[i] < o + len) { mapped[i] = part.removed ? n : n + (starts[i] - o); i++; }
    o += len;
    if (!part.removed) n += len;
  }
  while (i < starts.length) mapped[i++] = n;
  if (mapped.length) mapped[0] = 0;
  const regions: Region[] = [];
  const touched: string[] = [];
  last.regions.forEach((r, k) => {
    const s = mapped[k], e = k + 1 < mapped.length ? mapped[k + 1] : text.length;
    if (e <= s) { warnings.push(`degraded: ${r.id} empty, dropped until the next clean parse`); return; }
    const body = text.slice(s, e);
    const changed = body !== last.text.slice(r.start, r.end);
    if (changed) touched.push(r.id);
    regions.push({ ...r, start: s, end: e, hash: changed ? sha1(body) : r.hash, masked: changed ? sha1(body) : r.masked });
  });
  return { path: last.path, pack: last.pack, degraded: true, text, regions, warnings, touched };
}
```

```ts
// packages/indexer/src/regions/lifecycle.ts
import type { Region } from './types.js';

export interface RegionDiff { updated: string[]; created: string[]; tombstoned: string[]; renamed: Array<{ from: string; to: string }> }

/** §3.5. Renames need a one-to-one masked-hash match among gone and added regions of the same kind. Gaps follow their predecessor. */
export function diffRegions(old: Region[], neu: Region[]): RegionDiff {
  const o = new Map(old.map((r) => [r.id, r]));
  const n = new Map(neu.map((r) => [r.id, r]));
  const updated = neu.filter((r) => o.has(r.id) && o.get(r.id)!.hash !== r.hash).map((r) => r.id);
  const gone = old.filter((r) => !n.has(r.id));
  const added = neu.filter((r) => !o.has(r.id));
  const group = (rs: Region[]) => {
    const m = new Map<string, Region[]>();
    for (const r of rs) if (r.kind !== 'gap') (m.get(`${r.kind}|${r.masked}`) ?? m.set(`${r.kind}|${r.masked}`, []).get(`${r.kind}|${r.masked}`)!).push(r);
    return m;
  };
  const g = group(gone), a = group(added);
  const renamed: Array<{ from: string; to: string }> = [];
  const qnameTo = new Map<string, string>();
  for (const [k, rs] of g) {
    const cand = a.get(k);
    if (rs.length === 1 && cand?.length === 1) { renamed.push({ from: rs[0].id, to: cand[0].id }); qnameTo.set(rs[0].qname, cand[0].qname); }
  }
  for (const r of gone) {
    const m = /^gap@after:(.+)$/.exec(r.qname);
    if (r.kind !== 'gap' || !m || !qnameTo.has(m[1])) continue;
    const toId = r.id.replace(`::${r.qname}::`, `::gap@after:${qnameTo.get(m[1])}::`);
    if (n.has(toId) && !o.has(toId)) renamed.push({ from: r.id, to: toId });
  }
  const from = new Set(renamed.map((x) => x.from)), to = new Set(renamed.map((x) => x.to));
  return { updated, created: added.filter((r) => !to.has(r.id)).map((r) => r.id), tombstoned: gone.filter((r) => !from.has(r.id)).map((r) => r.id), renamed };
}
```

```ts
// packages/indexer/src/regions/refs.ts
import { posix } from 'node:path';
import type Parser from 'web-tree-sitter';
import { language, type Pack } from './pack.js';
import type { FileRegions } from './types.js';

type Node = Parser.SyntaxNode;
export interface RawRefs { calls: Array<{ at: number; name: string }>; imports: Array<{ at: number; local: string; module: string; name: string | null }> }

const queries = new Map<string, Parser.Query>();

export async function readRefs(pack: Pack, root: Node, path: string): Promise<RawRefs> {
  let q = queries.get(pack.name);
  if (!q) { q = (await language(pack.grammar)).query(pack.calls); queries.set(pack.name, q); }
  const calls = q.captures(root).filter((c) => c.name === 'callee').map((c) => ({ at: c.node.startIndex, name: c.node.text }));
  return { calls, imports: pack.importReader === 'python' ? pyImports(root, path) : esImports(root, path) };
}

function pyImports(root: Node, path: string): RawRefs['imports'] {
  const out: RawRefs['imports'] = [];
  const fileMod = path.replace(/\.pyi?$/, '').replace(/\/__init__$/, '').replaceAll('/', '.');
  const isPkg = /\/__init__\.pyi?$/.test(path);
  for (const st of root.namedChildren) {
    if (st.type === 'import_from_statement') {
      const mn = st.childForFieldName('module_name');
      if (!mn) continue;
      let mod = mn.text;
      if (mn.type === 'relative_import') {
        const dots = /^\.+/.exec(mod)![0].length;
        const rest = mod.slice(dots);
        const parts = fileMod.split('.');
        const base = isPkg ? parts : parts.slice(0, -1);
        mod = [...base.slice(0, base.length - (dots - 1)), ...(rest ? [rest] : [])].join('.');
      }
      for (const n of st.childrenForFieldName('name')) {
        const [name, local] = n.type === 'aliased_import' ? [n.childForFieldName('name')!.text, n.childForFieldName('alias')!.text] : [n.text, n.text];
        out.push({ at: st.startIndex, local, module: mod, name });
      }
    } else if (st.type === 'import_statement') {
      for (const n of st.childrenForFieldName('name')) {
        if (n.type === 'aliased_import') out.push({ at: st.startIndex, local: n.childForFieldName('alias')!.text, module: n.childForFieldName('name')!.text, name: null });
        else out.push({ at: st.startIndex, local: n.text.split('.')[0], module: n.text.split('.')[0], name: null });
      }
    }
  }
  return out;
}

function esImports(root: Node, path: string): RawRefs['imports'] {
  const out: RawRefs['imports'] = [];
  const dir = posix.dirname(path);
  for (const st of root.namedChildren) {
    if (st.type !== 'import_statement') continue;
    const src = st.childForFieldName('source');
    const clause = st.namedChildren.find((c) => c.type === 'import_clause');
    if (!src || !clause) continue;
    const spec = src.text.slice(1, -1);
    const module = spec.startsWith('.') ? posix.normalize(posix.join(dir, spec)).replace(/\.(d\.ts|[cm]?[jt]sx?)$/, '') : spec;

  }
```

```ts
export function buildEdges(files: FileRegions[]): { edges: Edge[]; unresolved: number } {

  const mods = new Map<string, string>();

    const m = new Map<string, string>();

      const last = r.qname.split('.').at(-1)!;

    }

    const qOf = new Map(f.regions.map((r) => [r.id, r.qname]));

      if (t) add({ kind: 'IMPORTS', from: b.from, to: t, via: 'import' });

      let via: Edge['via'] = 'same_file';
      if (DOTTED.test(c.name)) {
        const self = /^(self|this|cls)\.(.+)$/.exec(c.name);
        if (self) {
          const q = qOf.get(c.from) ?? '';

            const [head, ...rest] = c.name.split('.');
            const b = binds.get(head);
            if (b && (to = target(f, b, rest.join('.')))) via = 'import';
          }

        const cands = last ? byName.get(last) : undefined;
        if (cands?.length === 1) { to = cands[0]; via = 'unique_name'; }
      }
      if (!to) { unresolved++; continue; }
      add({ kind: 'CALLS', from: c.from, to, via });
      if (TEST_PATH.test(f.path)) add({ kind: 'TESTS', from: c.from, to, via });

      if (r.kind !== 'container') continue;

          add({ kind: 'CONTAINS', from: r.id, to: s.id, via: 'same_file' });
      }
    }
  }
  return { edges: [...out.values()], unresolved };
}
```

```ts
export type MergeResult =

  | { status: 'unsupported'; reason: string };

export function gitMergeFile(base: string, ours: string, theirs: string): { clean: boolean; text: string } {
  const d = mkdtempSync(join(tmpdir(), 'loom-merge-'));
  try {
    const [b, o, t] = ['base', 'ours', 'theirs'].map((n) => join(d, n));
    writeFileSync(b, base); writeFileSync(o, ours); writeFileSync(t, theirs);
    const r = spawnSync('git', ['merge-file', '-p', '-L', 'ours', '-L', 'base', '-L', 'theirs', o, b, t], { encoding: 'utf8' });
    if (r.status === null || r.status < 0) throw new Error(`git merge-file failed: ${r.stderr || r.error}`);
    return { clean: r.status === 0, text: r.stdout };
  } finally {
    rmSync(d, { recursive: true, force: true });
  }

const kindOf = (id: string) => id.slice(id.lastIndexOf('::') + 2).split('#')[0];

/** Split an imports region into statements (parenthesised and backslash continuations kept together). */
function statements(text: string): string[] {
  const out: string[] = [];
  let cur = '', depth = 0;
  for (const line of text.split(/(?<=\n)/)) {
    cur += line;
    for (const ch of line) depth += ch === '(' || ch === '{' ? 1 : ch === ')' || ch === '}' ? -1 : 0;
    if (depth <= 0 && !/\\\r?\n?$/.test(line)) { out.push(cur); cur = ''; depth = 0; }
  }
  if (cur) out.push(cur);
  return out;
}

function unionImports(base: string, ours: string, theirs: string): string {
  const key = (s: string) => s.trimEnd();
  const have = new Set([...statements(ours), ...statements(base)].map(key));
  const extra = statements(theirs).filter((s) => !have.has(key(s)));
  const o = ours.endsWith('\n') || !extra.length ? ours : `${ours}\n`;
  return o + extra.map((s) => (s.endsWith('\n') ? s : `${s}\n`)).join('');
}

/** §3.10. Tried when git's line merge conflicts; complements it rather than replacing it. */
export async function regionMerge(path: string, base: string, ours: string, theirs: string): Promise<MergeResult> {
  const [B, O, T] = await Promise.all([base, ours, theirs].map((s) => regionize(path, s)));
  for (const [side, fr] of [['base', B], ['ours', O], ['theirs', T]] as const) if (fr.degraded) return { status: 'unsupported', reason: `${side} does not parse` };
  const txt = (fr: typeof B) => new Map(fr.regions.map((r) => [r.id, fr.text.slice(r.start, r.end)]));
  const b = txt(B), o = txt(O), t = txt(T);
  const oIds = O.regions.map((r) => r.id), tIds = T.regions.map((r) => r.id);
  const common = (ids: string[], other: Map<string, string>) => ids.filter((id) => other.has(id)).join('\n');
  if (common(oIds, t) !== common(tIds, o)) return { status: 'unsupported', reason: 'region order differs between sides' };

  const conflicts: Array<{ id: string; kind: 'text' | 'delete' }> = [];
  const decide = (id: string): string | null => {
    const bb = b.get(id), oo = o.get(id), tt = t.get(id);
    if (oo !== undefined && tt !== undefined) {
      if (oo === tt) return oo;
      if (oo === bb) return tt;
      if (tt === bb) return oo;
      if (kindOf(id) === 'imports') return unionImports(bb ?? '', oo, tt);
      const m = gitMergeFile(bb ?? '', oo, tt);
      if (m.clean) return m.text;
      conflicts.push({ id, kind: 'text' });
      return null;
    }
    const one = oo ?? tt;
    if (one === undefined) return null;
    if (bb === undefined) return one;          // added on one side
    if (one === bb) return null;               // deleted on the other side, unchanged here
    conflicts.push({ id, kind: 'delete' });    // deleted vs modified
    return null;
  };

  const out = new Map<string, string | null>();
  for (const id of oIds) out.set(id, decide(id));
  const order = [...oIds];
  tIds.forEach((id, k) => {
    if (o.has(id)) return;
    const v = decide(id);
    if (v === null) return;
    out.set(id, v);
    let j = k - 1;
    while (j >= 0 && !order.includes(tIds[j])) j--;
    order.splice(j < 0 ? 0 : order.indexOf(tIds[j]) + 1, 0, id);
  });
  if (conflicts.length) return { status: 'conflict', regions: conflicts };
  const text = order.map((id) => out.get(id) ?? '').join('');
  const check = await regionize(path, text);
  if (check.degraded) return { status: 'unsupported', reason: 'merged text does not parse' };
  return { status: 'merged', text };
}
```

```ts
// packages/indexer/src/regions/lock.ts
import { createHash } from 'node:crypto';
import type { FileRegions, Region, RegionKind } from './types.js';

export interface LockRegion { id: string; kind: RegionKind; start: number; end: number; hash: string; m: string; sig: string | null }
export interface LockEntry { path: string; blob: string; pack: string; degraded: boolean; regions: LockRegion[] }

export function gitBlobSha(c: string | Buffer): string {
  const b = typeof c === 'string' ? Buffer.from(c, 'utf8') : c;
  return createHash('sha1').update(`blob ${b.length}\0`).update(b).digest('hex');
}

function sortedPass(offs: number[], step: (from: number, to: number) => number): number[] {
  const order = offs.map((o, i) => [o, i] as const).sort((a, b) => a[0] - b[0]);
  const out = new Array<number>(offs.length);
  let pos = 0, acc = 0;
  for (const [o, i] of order) { acc += step(pos, o); pos = o; out[i] = acc; }
  return out;
}

export const toUtf8Offsets = (text: string, offs: number[]): number[] => sortedPass(offs, (a, b) => Buffer.byteLength(text.slice(a, b)));

export function fromUtf8Offsets(text: string, offs: number[]): number[] {
  const buf = Buffer.from(text, 'utf8');
  return sortedPass(offs, (a, b) => buf.subarray(a, b).toString('utf8').length);
}

export function toLockEntry(fr: FileRegions, blob: string): LockEntry {
  const starts = fr.binary ? fr.regions.map((r) => r.start) : toUtf8Offsets(fr.text, fr.regions.map((r) => r.start));
  const ends = fr.binary ? fr.regions.map((r) => r.end) : toUtf8Offsets(fr.text, fr.regions.map((r) => r.end));
  return {
    path: fr.path, blob, pack: fr.pack, degraded: fr.degraded,
    regions: fr.regions.map((r, i) => ({ id: r.id, kind: r.kind, start: starts[i], end: ends[i], hash: r.hash, m: r.masked, sig: r.sig })),
  };
}

export function fromLockEntry(e: LockEntry, text: string): FileRegions {
  const starts = fromUtf8Offsets(text, e.regions.map((r) => r.start));
  const ends = fromUtf8Offsets(text, e.regions.map((r) => r.end));
  const regions: Region[] = e.regions.map((r, i) => ({
    id: r.id, kind: r.kind, qname: r.id.slice(e.path.length + 2, r.id.lastIndexOf('::')),
    start: starts[i], end: ends[i], hash: r.hash, masked: r.m, sig: r.sig,
  }));
  return { path: e.path, pack: e.pack, degraded: e.degraded, text, regions, warnings: [] };
}

export function serializeLock(entries: LockEntry[]): string {
  const one = new Map<string, LockEntry>();
  for (const e of entries) one.set(e.path, e);
  return [...one.values()].sort((a, b) => (a.path < b.path ? -1 : a.path > b.path ? 1 : 0))
    .map((e) => `${JSON.stringify({ path: e.path, blob: e.blob, pack: e.pack, degraded: e.degraded, regions: e.regions })}\n`).join('');
}

/** Tolerant: skips conflict markers, blank and malformed lines; keeps every entry per path (union merge can leave two). */
export function parseLock(text: string): Map<string, LockEntry[]> {
  const out = new Map<string, LockEntry[]>();
  for (const line of text.split('\n')) {
    if (!line.trim() || /^(<{7}|={7}|>{7}|\|{7})/.test(line)) continue;
    let e: LockEntry;
    try { e = JSON.parse(line); } catch { continue; }
    if (typeof e?.path !== 'string' || typeof e.blob !== 'string' || !Array.isArray(e.regions)) continue;
    (out.get(e.path) ?? out.set(e.path, []).get(e.path)!).push(e);
  }
  return out;
}

export const pickEntry = (entries: LockEntry[] | undefined, blob: string): LockEntry | undefined => entries?.find((e) => e.blob === blob);
```

```ts
// packages/cli/src/regions.ts
import { execFileSync } from 'node:child_process';
import { readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { checkInvariants, COMMENT_SYNTAX, regionize, type FileRegions } from '@loom/indexer';

const lineOf = (text: string, i: number) => { let n = 1; for (let k = 0; k < i; k++) if (text.charCodeAt(k) === 10) n++; return n; };

function render(fr: FileRegions): string[] {
  return fr.regions.map((r) => {
    const span = fr.binary ? 'bin' : `${lineOf(fr.text, r.start)}-${lineOf(fr.text, Math.max(r.start, r.end - 1))}`;
    return `${span.padEnd(7)}${r.kind.padEnd(12)}${r.qname}${r.sig ? `  ${r.sig}` : ''}`;
  });
}

export async function regionsCmd(rest: string[], cwd = process.cwd(), log: (s: string) => void = console.log): Promise<number> {
  const json = rest.includes('--json');
  const args = rest.filter((a) => a !== '--json');

  if (args[0] === 'check') {
    const files = args.length > 1 ? args.slice(1) : execFileSync('git', ['ls-files', '-z'], { cwd, encoding: 'utf8' }).split('\0').filter(Boolean);
    let regions = 0, degraded = 0;
    const problems: string[] = [];
    for (const f of files) {
      let buf: Buffer;
      try { buf = readFileSync(join(cwd, f)); } catch { continue; }
      const fr = await regionize(f, buf);
      regions += fr.regions.length;
      if (fr.degraded) degraded++;
      problems.push(...checkInvariants(fr).map((v) => `${f}: ${v}`));
    }
    const summary = { files: files.length, regions, degraded, violations: problems.length };
    log(json ? JSON.stringify({ ...summary, problems })
      : [...problems, `files ${summary.files} · regions ${regions} · degraded ${degraded} · violations ${problems.length}`].join('\n'));
    return problems.length ? 1 : 0;
  }

  if (args[0] === 'mark') {
    const [, file, s, e, name] = args;
    const syntax = file ? COMMENT_SYNTAX(file) : null;
    if (!file || !name || !/^\S+$/.test(name) || !syntax) { log('usage: loom regions mark <file> <start-line> <end-line> <name> (file needs line comments)'); return 2; }
    const text = readFileSync(join(cwd, file), 'utf8');
    const lines = text.split('\n');
    const last = text.endsWith('\n') ? lines.length - 1 : lines.length;
    const a = Number(s), b = Number(e);
    if (!(a >= 1 && b >= a && b <= last)) { log(`line range ${s}-${e} outside 1-${last}`); return 2; }
    const indent = /^[ \t]*/.exec(lines[a - 1])![0];
    const [open, close] = syntax;
    lines.splice(b, 0, `${indent}${open}loom:end${close}`);
    lines.splice(a - 1, 0, `${indent}${open}loom:region ${name}${close}`);
    const next = lines.join('\n');
    const fr = await regionize(file, next);
    if (!fr.regions.some((r) => r.kind === 'section' && r.qname === name)) { log(fr.warnings.join('\n') || `marker ${name} rejected`); return 1; }
    writeFileSync(join(cwd, file), next);
    log(`marked ${name} in ${file}`);
    return 0;
  }

  const [file] = args;
  if (!file) { log('usage: loom regions <file> | check [files…] | mark <file> <start-line> <end-line> <name>'); return 2; }
  const fr = await regionize(file, readFileSync(join(cwd, file)));
  if (json) log(JSON.stringify({ path: fr.path, pack: fr.pack, degraded: fr.degraded, warnings: fr.warnings, regions: fr.regions.map(({ masked: _m, ...r }) => r) }));
  else log([...render(fr), ...fr.warnings.map((w) => `! ${w}`), ...(fr.degraded ? ['! degraded: file does not parse'] : [])].join('\n'));
  return 0;
}
```
